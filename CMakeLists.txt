# =============================================================================
# Hover! - static recompilation host
#
#   py -3 run_lift.py        # -> src/recomp/gen/  (needs work/functions_hover.json)
#   build.cmd                # Windows: vcvarsall amd64_x86 + CMake + Ninja -> build/hover.exe
#   ./build.sh               # Linux/macOS: CMake + Ninja -> build/hover
#   build/hover --run
#
# Windows builds the lifted code on pcrecomp's runtime/native32 (MSVC x86, a
# 32-bit guest address is a 32-bit host address). Linux, macOS and Android
# build it on runtime/win32hle: the game's Win32 imports are implemented on
# libc/SDL2 (src/hle/main.c is the host there). With -DHOVER_GEN_STUB=ON the
# build uses an empty dispatch table and stub hosts, so CI can configure,
# compile and run the selftests without the game.
# =============================================================================
cmake_minimum_required(VERSION 3.20)
project(hover C)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# ../pcrecomp, or ../tools where older setups put it
if(NOT DEFINED PCRECOMP)
  if(EXISTS "${CMAKE_SOURCE_DIR}/../pcrecomp/runtime")
    set(PCRECOMP "${CMAKE_SOURCE_DIR}/../pcrecomp" CACHE PATH "pcrecomp checkout")
  else()
    set(PCRECOMP "${CMAKE_SOURCE_DIR}/../tools" CACHE PATH "pcrecomp checkout")
  endif()
endif()

# ---- the lifted code: src/recomp/gen from run_lift.py, or a stub for CI ----
option(HOVER_GEN_STUB "build with a stub dispatch table (no game in the tree)" OFF)
file(GLOB GEN_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/recomp/gen/*.c")
if(GEN_SOURCES)
  set(GEN_DIR "${CMAKE_SOURCE_DIR}/src/recomp/gen")
elseif(HOVER_GEN_STUB)
  set(GEN_DIR "${CMAKE_BINARY_DIR}/gen-stub")
  file(MAKE_DIRECTORY "${GEN_DIR}")
  file(WRITE "${GEN_DIR}/hover_stub.c"
"/* Generated by CMake (HOVER_GEN_STUB): an empty dispatch and the lifted
 * functions the host itself calls, so the host and its selftests build and
 * run without the game. Nothing here runs the game. */
#include <stdint.h>
#include \"recomp_types.h\"

const recomp_dispatch_entry_t recomp_dispatch_table[] = { { 0u, 0 } };
const uint32_t recomp_dispatch_count = 0u;
const uint32_t hover_entry_va = 0u;

void sub_00402250(void) {}
void sub_00405B90(void) {}
void sub_004089D0(void) {}
void sub_004089F0(void) {}
void sub_00409110(void) {}
void sub_00409180(void) {}
void sub_00409300(void) {}
void sub_0040A4C0(void) {}
void sub_0040A590(void) {}
void sub_0040CB40(void) {}
void sub_00414470(void) {}
void sub_004386D0(void) {}
")
  set(GEN_SOURCES "${GEN_DIR}/hover_stub.c")
  message(STATUS "src/recomp/gen is empty: building with the stub dispatch (HOVER_GEN_STUB)")
else()
  message(FATAL_ERROR "src/recomp/gen is empty -- run `py -3 run_lift.py` first, "
                      "or configure with -DHOVER_GEN_STUB=ON for a compile/selftest check")
endif()

set(GEN_OPT "-O2" CACHE STRING "optimisation for the generated C")
set_source_files_properties(${GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "-w;${GEN_OPT};-fno-strict-aliasing;-fwrapv")

# =============================================================================
# Linux, macOS, Android: the lifted C on runtime/win32hle
# =============================================================================
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR APPLE OR ANDROID)
  if(NOT EXISTS "${PCRECOMP}/runtime/win32hle/win32hle.h")
    message(FATAL_ERROR "no pcrecomp at ${PCRECOMP} (set -DPCRECOMP=...)")
  endif()
  if(ANDROID)
    # SDL2 and SDL2_ttf built for the target's ABI (the android job in
    # .github/workflows/ci.yml builds both from source into a prefix).
    # SDL_main.h renames src/hle/main.c's main() to SDL_main, which is the
    # symbol SDLActivity calls after loading libSDL2.so and libmain.so (the
    # bundler renames libhover.so).
    find_package(SDL2 CONFIG REQUIRED)
    find_package(SDL2_ttf CONFIG REQUIRED)
    set(HOVER_SDL2 SDL2::SDL2 SDL2_ttf::SDL2_ttf)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
    set(HOVER_SDL2 PkgConfig::SDL2 PkgConfig::SDL2_TTF)
  endif()
  set(HLE "${PCRECOMP}/runtime/win32hle")
  file(GLOB HLE_SOURCES CONFIGURE_DEPENDS "${HLE}/*.c")
  list(FILTER HLE_SOURCES EXCLUDE REGEX "_selftest\\.c$")
  set(HOVER_SOURCES
    src/hle/main.c
    src/port/port_posix.c
    src/port/pad_sdl.c
    src/port/net_offline.c
    src/runtime/mp.c
    src/runtime/levels.c
    ${HLE_SOURCES}
    ${GEN_SOURCES}
  )
  if(ANDROID)
    # Android app code is a shared library loaded by SDLActivity; the APK
    # bundler (packaging/bundle-android.sh) renames libhover.so to libmain.so.
    add_library(hover SHARED ${HOVER_SOURCES})
  else()
    add_executable(hover ${HOVER_SOURCES})
  endif()
  target_include_directories(hover PRIVATE
    src/port src/runtime ${GEN_DIR} ${HLE} ${PCRECOMP}/runtime/recomp32)
  target_compile_definitions(hover PRIVATE RECOMP_RETADDR=0u RECOMP_LOCAL_REGS)
  # This host's own recomp_lookup_manual (mp.c's seats) takes the place of
  # win32hle's stub: rename the stub at compile time.
  set_source_files_properties("${HLE}/host_lite.c"
    PROPERTIES COMPILE_DEFINITIONS "recomp_lookup_manual=win32hle_lookup_manual_unused")
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_compile_definitions(hover PRIVATE RECOMP_FLAT_MEMORY)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_link_options(hover PRIVATE -no-pie)   # the guest's 0x00400000 stays free
    endif()
  else()
    # 64-bit: where the guest's 4 GB goes in the host. A fixed address lets
    # the lifted code add a constant, not a global it reloads after every
    # store. On a Mac 16 TB up is free on arm64 and x86-64 alike; on Linux
    # the default is wherever mmap puts it. Empty: wherever mmap puts it.
    if(APPLE)
      set(HOVER_MEM_BASE "0x100000000000ull" CACHE STRING "host address of the guest's 4 GB (empty: anywhere)")
    else()
      set(HOVER_MEM_BASE "" CACHE STRING "host address of the guest's 4 GB (empty: anywhere)")
    endif()
    if(HOVER_MEM_BASE)
      target_compile_definitions(hover PRIVATE RECOMP_MEM_BASE=${HOVER_MEM_BASE})
    endif()
  endif()
  target_compile_options(hover PRIVATE -fno-strict-aliasing -fwrapv)
  # Threads::Threads, not a literal pthread: glibc wants -pthread/-lpthread,
  # but Bionic has the pthread symbols in libc and ships no -lpthread at all
  # (the Android link failed on "unable to find library -lpthread").
  find_package(Threads REQUIRED)
  target_link_libraries(hover PRIVATE ${HOVER_SDL2} Threads::Threads m)
  if(ANDROID)
    target_link_libraries(hover PRIVATE log)
  endif()
  return()
endif()

# =============================================================================
# Windows: the host on runtime/native32 (32-bit, MSVC x86)
# =============================================================================
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "the Windows host must be 32-bit: use build.cmd (vcvarsall amd64_x86)")
endif()
if(NOT MSVC)
  message(FATAL_ERROR "runtime/native32 is MSVC x86 inline assembly")
endif()

add_executable(hover
  src/runtime/host.c
  src/runtime/pad.c
  src/runtime/present.c
  src/runtime/levels.c
  src/runtime/mp.c
  src/runtime/net.c
  ${PCRECOMP}/runtime/native32/native32.c
  ${PCRECOMP}/runtime/recomp32/image_loader.c
  ${PCRECOMP}/runtime/recomp32/recomp_trace.c
  ${GEN_SOURCES}
)
target_include_directories(hover PRIVATE
  src/port src/runtime ${GEN_DIR} ${PCRECOMP}/runtime/native32 ${PCRECOMP}/runtime/recomp32)
# RECOMP_RETADDR=0u: a stack imbalance during bring-up leaks a harmless 0, not
# 0xDEAD0000 into a size argument (forcecommander).
target_compile_definitions(hover PRIVATE RECOMP_RETADDR=0u _CRT_SECURE_NO_WARNINGS
                                         RECOMP_FLAT_MEMORY RECOMP_LOCAL_REGS)
# Linked high and not relocatable, so HOVER.EXE's 0x00400000 is free for the
# guest image.
target_link_options(hover PRIVATE /BASE:0x60000000 /FIXED:NO /DYNAMICBASE:NO
                                  /NXCOMPAT /STACK:0x1000000 /MANIFESTUAC:NO)
set_source_files_properties(${GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "/w;/O2")
set_source_files_properties(src/runtime/host.c src/runtime/pad.c src/runtime/present.c
                            src/runtime/levels.c src/runtime/mp.c src/runtime/net.c
                            PROPERTIES COMPILE_OPTIONS "/W3")
target_link_libraries(hover PRIVATE user32 gdi32 kernel32 advapi32 winmm dwmapi shell32 d3d11 dxgi dxguid ws2_32)

option(HOVER_TRACE "record a ring buffer of lifted function entries" OFF)
if(HOVER_TRACE)
  target_compile_definitions(hover PRIVATE RECOMP_TRACE=1)
endif()
